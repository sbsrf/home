// 根据选择的方案生成 data 数组
const data = computed(() => {
  const scheme = schemes[selectedScheme.value];
  // 无理字等练习直接提供 [内容, 编码] 数据对
  if (scheme.data) {
    return scheme.data;
  }
  const result = [];
  const keymap = scheme.keymap;
  Object.entries(keymap).map(([key, group]) => {
    Array.from(group).forEach((radical) => {
      result.push([radical, key]);
    });
  });
  return result;
});
